import React, { useState } from 'react';
import { useQuery } from '@tanstack/react-query';
import { apiRequest } from '@/lib/apiRequest';
import { Input } from '@/components/ui/input';
import { Button } from '@/components/ui/button';
import { Select, SelectItem } from '@/components/ui/select';
import { Card } from '@/components/ui/card';
import { Checkbox } from '@/components/ui/checkbox';
import { Accordion, AccordionItem } from '@/components/ui/accordion';
import { Tabs, TabsList, TabsTrigger } from '@/components/ui/tabs';

const itemTypes = ['task', 'habit', 'goal', 'project', 'chore'];

export default function Plans() {
  const [statusFilter, setStatusFilter] = useState('incomplete');
  const [typeFilter, setTypeFilter] = useState('all');
  const [searchTerm, setSearchTerm] = useState('');
  const [showRecurring, setShowRecurring] = useState(false);

  const { data: items = [] } = useQuery({
    queryKey: ['plans', statusFilter, typeFilter],
    queryFn: () => apiRequest(`/api/items?status=${statusFilter}`)
  });

  const filteredItems = items.filter(item => {
    const matchesType = typeFilter === 'all' || item.type === typeFilter;
    const matchesSearch = item.title.toLowerCase().includes(searchTerm.toLowerCase());
    return matchesType && matchesSearch;
  });

  const pastDueItems = filteredItems.filter(item => item.due_date && new Date(item.due_date) < new Date());
  const recurringItems = filteredItems.filter(item => item.recurrence);
  const nonRecurringItems = filteredItems.filter(item => !item.recurrence);

  return (
    <div className="max-w-2xl mx-auto p-4 space-y-4">
      <div className="sticky top-0 z-10 bg-background py-2">
        <h1 className="text-2xl font-bold">Your Plans</h1>
        <p className="text-sm text-muted-foreground">Everything you’ve created — organized and filterable</p>
      </div>

      <div className="flex flex-col gap-2 md:flex-row md:items-center md:justify-between">
        <Input
          placeholder="Search your items..."
          value={searchTerm}
          onChange={(e) => setSearchTerm(e.target.value)}
        />

        <div className="flex gap-2">
          <Select value={typeFilter} onValueChange={setTypeFilter}>
            <SelectItem value="all">All Types</SelectItem>
            {itemTypes.map(type => (
              <SelectItem key={type} value={type}>{type}</SelectItem>
            ))}
          </Select>

          <Select value={statusFilter} onValueChange={setStatusFilter}>
            <SelectItem value="incomplete">Not Completed</SelectItem>
            <SelectItem value="complete">Completed</SelectItem>
          </Select>
        </div>
      </div>

      {pastDueItems.length > 0 && (
        <Accordion type="single" collapsible>
          <AccordionItem value="pastDue">
            <div className="text-sm font-medium text-red-600">⚠ Past Due ({pastDueItems.length})</div>
            <div className="space-y-2 pt-2">
              {pastDueItems.map(item => (
                <Card key={item.id} className="p-3">
                  <p className="font-medium">{item.title}</p>
                  <p className="text-xs text-muted-foreground">Was due on {item.due_date}</p>
                </Card>
              ))}
            </div>
          </AccordionItem>
        </Accordion>
      )}

      <Tabs defaultValue="current">
        <TabsList>
          <TabsTrigger value="current">Current</TabsTrigger>
          <TabsTrigger value="recurring" onClick={() => setShowRecurring(true)}>Recurring</TabsTrigger>
        </TabsList>

        <div className="space-y-3 pt-4">
          {(showRecurring ? recurringItems : nonRecurringItems).map(item => (
            <Card key={item.id} className="p-4 flex justify-between items-center">
              <div>
                <p className="font-semibold">{item.title}</p>
                <p className="text-xs text-muted-foreground">
                  {item.type} {item.due_date && `• Due ${item.due_date}`} {item.recurrence && `• ${item.recurrence}`}
                </p>
              </div>
              <Checkbox checked={item.status === 'complete'} disabled />
            </Card>
          ))}
        </div>
      </Tabs>

      <div className="fixed bottom-6 right-6">
        <Button size="icon" onClick={() => window.location.href = '/create/ai'}>
          +
        </Button>
      </div>
    </div>
  );
}
