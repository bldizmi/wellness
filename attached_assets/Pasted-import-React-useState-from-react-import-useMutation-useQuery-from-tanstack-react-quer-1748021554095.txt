import React, { useState } from 'react';
import { useMutation, useQuery } from '@tanstack/react-query';
import { apiRequest } from '@/lib/apiRequest';
import { Input } from '@/components/ui/input';
import { Button } from '@/components/ui/button';
import { Textarea } from '@/components/ui/textarea';
import { Card } from '@/components/ui/card';
import { Select, SelectItem } from '@/components/ui/select';
import { Checkbox } from '@/components/ui/checkbox';
import { useNavigate } from 'react-router-dom';

export default function CreateWithAI() {
  const navigate = useNavigate();
  const [prompt, setPrompt] = useState('');
  const [structuredItem, setStructuredItem] = useState(null);

  const { data: profile } = useQuery({
    queryKey: ['profile'],
    queryFn: () => apiRequest('/api/profile')
  });

  const { data: communityMembers } = useQuery({
    queryKey: ['community-members'],
    queryFn: () => apiRequest('/api/community/default/members?excludeSelf=true')
  });

  const generate = useMutation({
    mutationFn: () => apiRequest('/api/ai/create-item-from-prompt', {
      method: 'POST',
      body: JSON.stringify({ input: prompt, profile })
    }),
    onSuccess: (data) => setStructuredItem(data)
  });

  const save = useMutation({
    mutationFn: () => apiRequest('/api/item', {
      method: 'POST',
      body: JSON.stringify(structuredItem)
    }),
    onSuccess: () => navigate('/shared')
  });

  const handleUpdate = (field, value) => {
    setStructuredItem(prev => ({ ...prev, [field]: value }));
  };

  const toggleShare = (userId) => {
    const current = structuredItem.shared_with || [];
    const updated = current.includes(userId)
      ? current.filter(id => id !== userId)
      : [...current, userId];
    handleUpdate('shared_with', updated);
  };

  return (
    <div className="relative max-w-xl mx-auto p-4 space-y-4">
      <div className="fixed bottom-6 right-6 z-50">
        <Button onClick={() => navigate('/create/ai')} variant="default">
          + New Task with AI
        </Button>
      </div>

      <h1 className="text-xl font-bold">Create with AI</h1>
      <Textarea
        placeholder="Describe your task or chore in natural language..."
        value={prompt}
        onChange={(e) => setPrompt(e.target.value)}
      />
      <Button onClick={() => generate.mutate()} disabled={generate.isPending}>
        {generate.isPending ? 'Generating...' : 'Generate Task'}
      </Button>

      {structuredItem && (
        <Card className="p-4 space-y-3">
          <Input
            placeholder="Title"
            value={structuredItem.title || ''}
            onChange={(e) => handleUpdate('title', e.target.value)}
          />

          <Select label="Type" value={structuredItem.type || ''} onValueChange={(val) => handleUpdate('type', val)}>
            {['task', 'habit', 'goal', 'project', 'chore'].map(t => (
              <SelectItem key={t} value={t}>{t}</SelectItem>
            ))}
          </Select>

          <Input
            placeholder="Due Date (YYYY-MM-DD)"
            value={structuredItem.due_date || ''}
            onChange={(e) => handleUpdate('due_date', e.target.value)}
          />

          <Input
            placeholder="Estimated Time (minutes)"
            type="number"
            value={structuredItem.estimated_time || ''}
            onChange={(e) => handleUpdate('estimated_time', e.target.value)}
          />

          <Textarea
            placeholder="Why is this important?"
            value={structuredItem.why_important || ''}
            onChange={(e) => handleUpdate('why_important', e.target.value)}
          />

          <div>
            <p className="text-sm font-medium">Share With</p>
            <div className="flex flex-wrap gap-2 mt-1">
              {communityMembers?.map(user => (
                <Checkbox
                  key={user.user_id}
                  label={user.name}
                  checked={structuredItem.shared_with?.includes(user.user_id) || false}
                  onCheckedChange={() => toggleShare(user.user_id)}
                />
              ))}
            </div>
          </div>

          <Button onClick={() => save.mutate()} disabled={save.isPending}>
            {save.isPending ? 'Saving...' : 'Save Item'}
          </Button>
        </Card>
      )}

      <p className="text-sm text-muted-foreground text-center">
        or{' '}
        <button className="underline" onClick={() => navigate('/create')}>
          edit an existing task manually
        </button>
      </p>
    </div>
  );
}
